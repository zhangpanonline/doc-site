os / os.path 高频函数

re 标题处理

import re

HEADING_RE = re.compile(r"^(#{1,6})\s+(.+)$", re.MULTILINE)
# 组1 = 井号（1-6 个 → 标题级数），组2 = 标题内容

def demote(match):
    hashes, content = match.group(1), match.group(2)
    level = len(hashes)
    if level < 6:
        return f'{"#" * (level + 1)} {content}'   # 降一级
    return f"**{content}**"                        # 六级 → 加粗正文

result = HEADING_RE.sub(demote, text)
