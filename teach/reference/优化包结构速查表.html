配置与调试速记

# pydantic-settings
class WebSettings(BaseSettings):
    app_name: str = "Web Service API"
    model_config = {
        "env_file": ".env",     # 读取来源（不入库）
        "env_prefix": "WEB_",   # 字段 → WEB_APP_NAME
    }

# debugpy 调试服务器
python -m debugpy --listen 0.0.0.0:5678 --wait-for-client \
    -m fastapi dev apps/web-service/app/main.py --port 8000
# VSCode: attach 模式连 localhost:5678
.env vs .env.example
.env 真实值 + 进 .gitignore；.env.example 模板进仓库。
env_prefix
字段名自动加前缀映射环境变量（app_name → WEB_APP_NAME）。
